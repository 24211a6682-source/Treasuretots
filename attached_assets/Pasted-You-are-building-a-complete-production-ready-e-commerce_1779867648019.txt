Backend — backend/payments/razorpay.py:

import razorpay, hmac, hashlib, os

client = razorpay.Client(auth=(
    os.getenv("RAZORPAY_KEY_ID"),
    os.getenv("RAZORPAY_KEY_SECRET")
))

def create_order(amount_rupees, receipt):
    return client.order.create({
        "amount": int(amount_rupees * 100),
        "currency": "INR",
        "receipt": receipt,
        "payment_capture": 1
    })

def verify_signature(order_id, payment_id, signature):
    msg = f"{order_id}|{payment_id}"
    expected = hmac.new(
        os.getenv("RAZORPAY_KEY_SECRET").encode(),
        msg.encode(), hashlib.sha256
    ).hexdigest()
    return hmac.compare_digest(expected, signature)

def verify_webhook(raw_body, signature):
    expected = hmac.new(
        os.getenv("RAZORPAY_WEBHOOK_SECRET").encode(),
        raw_body, hashlib.sha256
    ).hexdigest()
    return hmac.compare_digest(expected, signature)

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE users (
  id            UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name          VARCHAR(100) NOT NULL,
  email         VARCHAR(255) UNIQUE,
  phone         VARCHAR(15) UNIQUE,
  password_hash TEXT NOT NULL,
  role          VARCHAR(20) DEFAULT 'user' CHECK (role IN ('user','admin')),
  created_at    TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE products (
  id           UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name         VARCHAR(200) NOT NULL,
  description  TEXT,
  price        NUMERIC(10,2),
  stock        INTEGER DEFAULT 999,
  cover_image  TEXT,
  images       JSONB,
  category     VARCHAR(100),
  subcategory  VARCHAR(100),
  slug         VARCHAR(200) UNIQUE,
  is_buyable   BOOLEAN DEFAULT TRUE,
  is_active    BOOLEAN DEFAULT TRUE,
  created_at   TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE orders (
  id                UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id           UUID REFERENCES users(id),
  total_amount      NUMERIC(10,2),
  payment_status    VARCHAR(30) DEFAULT 'pending'
                    CHECK (payment_status IN ('pending','paid','failed','refunded')),
  order_status      VARCHAR(50) DEFAULT 'order_received'
                    CHECK (order_status IN ('order_received','making','dispatched','delivered')),
  child_name        VARCHAR(100),
  shipping_address  JSONB,
  razorpay_order_id TEXT,
  created_at        TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE order_items (
  id         UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  order_id   UUID REFERENCES orders(id) ON DELETE CASCADE,
  product_id UUID REFERENCES products(id),
  quantity   INTEGER DEFAULT 1,
  price      NUMERIC(10,2)
);

CREATE TABLE payments (
  id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  order_id            UUID REFERENCES orders(id),
  razorpay_payment_id TEXT UNIQUE,
  razorpay_order_id   TEXT,
  razorpay_signature  TEXT,
  status              VARCHAR(30) DEFAULT 'pending',
  amount              NUMERIC(10,2),
  paid_at             TIMESTAMPTZ,
  created_at          TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_products_category ON products(category);
CREATE INDEX idx_products_slug ON products(slug);
CREATE INDEX idx_orders_user_id ON orders(user_id);
CREATE INDEX idx_orders_payment_status ON orders(payment_status);
